function isNumberKey(evt) {
    var charCode = (evt.which) ? evt.which : event.keyCode
    if (charCode > 31 && (charCode < 48 || charCode > 57) && (charCode < 45 || charCode > 48))
        return false;

    return true;
}

function CheckIfLibrarySelected()
{

function GetRatingNumberOfAStar(starNumber, totalRating) {
    var RatingNumber = 0;

    if (starNumber == 1 && totalRating < 1)
    { RatingNumber = parseInt(totalRating); }
    else if (totalRating >= starNumber)
    { RatingNumber = 100; }
    else if (totalRating > (starNumber - 1)) {
        RatingNumber = (totalRating % (starNumber - 1)) * 100;
        RatingNumber = parseInt(RatingNumber);
    }

    return RatingNumber;
}

function SetRatingOfTitle(titleType, totalRating) {

    var Stars2 = $("#divAvgRating2 .avgRatings");

    for (var i = 0; i < Stars1.length; i++) {
        var width = GetRatingNumberOfAStar((i + 1), totalRating)

        $(Stars1[i]).css("width", width + "%");
        $(Stars2[i]).css("width", width + "%");
    }
}

function SetTitleRatingCount(TitleNo, TitleType) {
    $(".loader-area, .loader").css("display", "block");

    var model = {
        TitleNo: TitleNo,
        LibId: $('#txtSearchLibrary').attr("data-libid"),
        TitleType: TitleType
    };

    $.ajax({
        url: "/OPAC_V3/GetTitleRatingPercentCount/",
        data: model,
        type: "GET",
        success: function (data) {
            $(".loader-area, .loader").fadeOut('slow');
            // Check for SQL injection error code -- ADDED BY SONALI ON [28 JULY 2025]
            if (data === -100) {
                toastr.error('Invalid input detected. Please avoid special characters and SQL keywords.', 'Security Warning');
                return;
            } // END BY SONALI 

            var totalRatingCount = 0;
            var numOfStar = 5;  // Number of Rating Stars for showing percent progress bar.
            var starsRatCnt = [];

            for (var i = 0 ; i < numOfStar; i++) {
                starsRatCnt[i] = 0;
            }

            if (data != null && data != "-99") {

                if (data.length > 0) {
                    $.each(data, function (key, item) {

                        totalRatingCount += item.Count;

                        for (var i = 0 ; i < numOfStar; i++) {
                            if (item.Rating > i && item.Rating <= (i + 1))
                            { starsRatCnt[i] += item.Count }
                        }

                    });
                }

            }
            else {
                toastr.warning("Sorry, unable to load Rating Percents!");
            }

            if (totalRatingCount == 0) { totalRatingCount = 1; }

            for (var i = 0 ; i < numOfStar ; i++) {
                var width = (starsRatCnt[i] / totalRatingCount) * 100;
                width = width.toFixed(2);
                width = width + "%";
                $("#divProgressStar" + (i + 1)).css("width", width);
                $("#spProgressbarPercentStar" + (i + 1)).text(width);
            }

        },
        error: function () {
            toastr.error('Sorry, unable to load Rating Percents!');
            $(".loader-area, .loader").fadeOut('slow');

        }
    });
}

function SaveFeedbackRatingLikes(Titleno, TitleType, FRL, Input) // FRL : F = FEEDBACK, R = RATING, L = LIKE
{
    /*  TitleType:  Book, EBook, Journal, Thesis, Blog */
    var RATING = 0.00;
    var FEEDBACK = "";
    var LIKE = false;

    if (FRL == "FEEDBACK") { FEEDBACK = Input; }
    else if (FRL == "RATING") { RATING = Input; }
    else if (FRL == "LIKE") { LIKE = Input; }

    var postData = { PARAMETER: FRL, TITLENO: Titleno, LIBRARYID: $('#txtSearchLibrary').attr("data-libid"), Type: TitleType, RATING: RATING, FEEDBACK: FEEDBACK, LIKE: LIKE }

    $(".loader-area, .loader").css("display", "block");

    $.ajax({
        url: "/OPAC_V3/SaveFeedbackRatingLikes",
        data: JSON.stringify(postData),
        type: "POST",
        //async: false,
        contentType: "application/json;charset=utf-8",
        success: function (data) {

            $(".loader-area, .loader").fadeOut('slow');
            // Check for SQL injection error code --  ADDED BY SONALI ON [28 JULY 2025]
            if (data === -100) {
                toastr.error('Invalid input detected. Please avoid special characters and SQL keywords.', 'Security Warning');
                return;
            } // END BY SONALI 

            if (data == null || data == -99 || data.IsErrorInService == true) {
                toastr.error('Sorry, there is some error occured while processing your request! Please try again or contact to Administrator!');
            }
            else if (data == -1) {
                toastr.warning("Sorry !!! Please Login to give Feedback / Likes / Rating.");
            }
            else {
                
                
                if (FRL == "LIKE")
                {
                    if (TitleType == "Blog") {                        
                        $('#lblLikeCountOfBlog_' + Titleno).text(data.TotalBookLike);
                        $('#lblCommentCountOfBlog_' + Titleno).text(data.TotalFeedback);
                        $("#spnLikesCount").text(data.TotalBookLike);
                        $("#spnCommentsCount").text(data.TotalFeedback);
                    }
                }
                else if (FRL == "RATING") {
                    //toastr.success('Your Ratings has saved successfully.', 'Success..!');
                    var tName = TitleType == "Thesis" ? "Dissertation" : TitleType;
                    SetRatingOfTitle(tName, data.TotalRatingback);
                    SetTitleRatingCount(Titleno, TitleType)
                }
                else if (FRL == "FEEDBACK") {
                    toastr.success('Your Feedback / Comment has saved successfully.', 'Success..!');
                    GetReviews(Titleno, TitleType);
                    if (TitleType == "Blog") {

function MakeDemandForBook(SERIESCODE, ACCNO) {

    var txtDemandForDate = $("#txtDemandForDate_" + SERIESCODE + ACCNO);

    if (txtDemandForDate.val() == "") {
        toastr.warning("Please select a date on which you want to issue this book.");
        txtDemandForDate.focus();
        return false;
    }

    //var DemandForDate = txtDemandForDate.val().split("-");
    //var ddMMyyyy = DemandForDate[2] + "-" + DemandForDate[1] + "-" + DemandForDate[0] ; 

    var model = {
        Accno: ACCNO,
        SERIESCODE: SERIESCODE,
        //Btid: $('#hdnusername').val(),
        Library_ID: $('#txtSearchLibrary').attr("data-libid"),
        Demanddate: txtDemandForDate.val() //ddMMyyyy
    };

    $.ajax({
        url: "/OPAC_V3/SaveDemandOfBook/",
        data: JSON.stringify(model),
        type: "POST",
        contentType: "application/json;charset=utf-8",
        success: function (Data) {
            $(".loader-area, .loader").fadeOut('slow');

            if (Data == null) {
                toastr.warning("Something Went Wrong");
            }
            if (Data.status == -1) {
                toastr.error("Borrower has not logged in! Please first do login to OPAC then try again.");
            }
            else if (Data.status == 1) {
                toastr.success("DEAMND CREATED SUCCESSFULLY!");

                $.ajax({
                    url: "/OPAC_V3/SendDemandEmail/",
                    data: JSON.stringify(model),
                    type: "POST",
                    contentType: "application/json;charset=utf-8",
                    success: function (data) {
                        if (data == -99) {
                            toastr.warning("Sorry, some error has occured while sending book demand email to Borrower!");
                        }
                        else if (data == -1 || data == -3) {
                            toastr.warning("Sorry, configuration data is not available to send Book Demand Email/SMS! Please Contact to Administrator.");
                            return false;
                        }
                        else if (data == 1) {
                            toastr.success("Email has sent to Borrower!");
                        }
                        else if (data == 2) {
                            toastr.warning("Email has not sent to Borrower!");
                        }
                        else if (data == -4) {
                            toastr.warning("Borrower email not found to send email!");
                        }

                    },
                    error: function () {
                        toastr.error('Exception Occurred !');

                    }
                });

                $.ajax({
                    url: "/OPAC_V3/SendDemandSMS/",
                    data: JSON.stringify(model),
                    type: "POST",
                    contentType: "application/json;charset=utf-8",
                    success: function (data) {
                        if (data == -99) {
                            toastr.warning("Sorry, some error has occured while sending book demand SMS to Borrower!");
                        }
                        else if (data == -1 || data == -3) {
                            toastr.warning("Sorry, configuration data is not available to send Book Demand Email/SMS! Please Contact to Administrator.");
                            return false;
                        }
                        else if (data == 1) {
                            toastr.success("SMS has sent to Borrower!");
                        }
                        else if (data == 2) {
                            toastr.warning("SMS has not sent to Borrower!");
                        }
                        else if (data == -4) {
                            toastr.warning("Borrower's mobile number not found to send SMS!");
                        }
                    },
                    error: function () {
                        toastr.error('Exception Occurred !');
                    }
                });

            }
            else if (Data.status == 2) {
                toastr.warning("Sorry, This Book Is Already Demanded By You!");
            }
            else if (Data.status == 3) {
                toastr.warning("Sorry, This Book is Already Demanded By Other User!");
            }
            else if (Data.status == 4) {
                toastr.warning("Sorry, This Book Is Under Reservation!");
            }
            else if (Data.status == 5) {
                toastr.warning("Sorry, Your BT Valid Date Has Expired!");
            }
            else if (Data.status == 6) {
                toastr.warning("Sorry, Your Book Issue Limit Is Crossed!");
            }
            else if (Data.status == 7) {
                toastr.warning("Sorry, This Book Is Already Issued By You!");
            }
            else if (Data.status == 8) {
                toastr.warning("Sorry, This Book Is Already Demanded By You!");
            }
            else if (Data.status == 9) {
                toastr.warning("Sorry, Maximum Limit For Demands For Selected Date Has Already Exceed! You can Make Demand For Another Date.");
            }
            else if (Data.status == 10) {
                toastr.warning("Sorry, This Book Is Not Present!");
            }
            else if (Data.status == 11) {
                toastr.warning("Sorry, Past Date For Demand Is Not Allowed! Kindly Enter Present/Future Date");
                txtDemandForDate.focus();
            }

        },
        error: function () {
            toastr.error('Exception Occurred !');
            $(".loader-area, .loader").fadeOut('slow');

        }
    });
}

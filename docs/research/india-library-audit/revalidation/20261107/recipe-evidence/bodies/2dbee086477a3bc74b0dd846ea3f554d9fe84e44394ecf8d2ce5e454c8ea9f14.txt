function SearchTabChange()
{
    $(".forBookSearch").hide();

    $("#divBookSearchBy").hide();
    $("#divJournalSearchBy").hide();
    $("#divDissertationSearchBy").hide();
    $("#divEbookSearhBy").hide();
    $("#btnOpenAdvSearch").hide();
    $("#divESubscriptionSearchBy").hide();

    $("#divSearchResult").html("");

    switch (contentSearching) {
        case "Books":
            $(".forBookSearch").show();
            $("#divBookSearchBy").show();
            $("#btnOpenAdvSearch").show();
            break;

        case "Journal":
            $("#divJournalSearchBy").show();
            break;

        case "Dissertation":
            $("#divDissertationSearchBy").show()
            break;

        case "Ebook":
            $("#divEbookSearhBy").show();
            break;
//Added by Rahul Kahate 06-10-2022
        case "EBookSubscription":
            $("#divESubscriptionSearchBy").show();
            $("#nav-tabContent").hide();
            $("#btnSearch").hide();
            $("#clearSearch").hide();

            SearchEBookSubscription();
            break;
            // End by Rahul Kahate 06-10-2022
    }

    GetTopAndNewArrivalBooks(contentSearching);
    $("#spnNewArrivalHead").text(contentSearching);
    $("#spnTopBookHead").text(contentSearching);

}

$('.btnSearch').click(function () {

    if ($("#txtSearchText").val().trim() == "") {
        toastr.warning("Please Enter Text To Search");
        $('#txtSearchText').focus();
        return false;
    }

    // ADDED BY SONALI ON [30 APRIL 2026] - TID 50870 - VAPT XSS Protection - Client Side Validation
    var searchText = $("#txtSearchText").val().trim();

    // Step 1: Check for dangerous special characters
    // Allow only: letters, numbers, spaces, and comma
    // Block: hyphen, apostrophe, period, parentheses, and all special characters
    var specialCharPattern = /[!@#$%^&*_+=\[\]{};:"\\|<>?\/~\'.()]/;
    if (specialCharPattern.test(searchText)) {
        toastr.error("Special characters are not allowed. Only letters, numbers, spaces, and commas are permitted.");
        $('#txtSearchText').focus();
        return false;
    }

    // Step 2: Check for dangerous XSS patterns
    var dangerousPatterns = [
        /<script/i,
        /<\/script>/i,
        /javascript:/i,
        /onerror/i,
        /onload/i,
        /onclick/i,
        /onmouseover/i,
        /<iframe/i,
        /<embed/i,
        /<object/i,
        /eval\s*\(/i,
        /alert\s*\(/i,
        /document\./i,
        /window\./i,
        /<img/i,
        /<svg/i,
        /<style/i
    ];

    var isInvalid = false;
    for (var i = 0; i < dangerousPatterns.length; i++) {
        if (dangerousPatterns[i].test(searchText)) {
            isInvalid = true;
            break;
        }
    }

    if (isInvalid) {
        toastr.error("Invalid input detected. HTML tags and script elements are not allowed.");
        $('#txtSearchText').focus();
        return false;
    }
    // END BY SONALI

    if (!CheckIfLibrarySelected()) {        
        return false;
    }
    
    switch (contentSearching)
    {
        case "Books":
            SearchBooks();
            break;

        case "Journal":
            SearchJournal();
            break;

        case "Dissertation":
            SearchDissertation();
            break;

        case "Ebook":
            SearchEbooks();
            break;

        //case "EBookSubscription":
        //    SearchEBookSubscription();
        //    break;

    }
    
});

function GetJournalTitleDetails(TitleNo) {
    
    var TitleType = "Journal";
  
    AttachFeedbackRatingDiv(TitleType);
    AttachAvgRatingDiv(TitleType);
    $("#btnSubmitReview").attr("data-titleType", TitleType);

    $(".loader-area, .loader").css("display", "block");
    
    $.ajax({
        url: "/OPAC_V3/GetJournalDetails?TitleNo=" + TitleNo + "&LibId=" + $('#txtSearchLibrary').attr("data-libid"),
        type: "GET",
        dataType: 'json',
        success: function (data) {
                if (data != null && data != -99) {
                //$("#reserveBook").hide();
                //$("#demandBook").hide();

                //if (AllowBookReservation)
                //{ $("#reserveBook").show(); }

                //if (AllowBookDemand)
                //{ $("#demandBook").show(); }

                $("#presentJournalCount").text(data.TOTAL_COPIES);
                $("#issuedJournalCount").text(data.ISSUED_COPIES);

                $("#imgJournalTitle").attr("src", $("#imgOfTitle" + TitleNo).attr("src"))

                $("#spanJournalTitleName").text(data.JOURTITLENAME);
                $("#spJournalTitle").text(data.JOURTITLENAME);
                $("#spJournalTitle").attr("data-titleId", TitleNo);
                $("#spJournalCategory").text(data.CATEGOURY);
                $("#spJournalType").text(data.JOURTITLETYPE);
                $("#spJournalDepartment").text(data.DEPTNAME);
                $("#spJournalPublisher").text(data.PUBNAME);
                $("#spJournalFrequency").text(data.FREQUENCY);
                $("#spJournalISSN").text(data.ISSN);
                

                var html = "";
                $.each(data.JournalSerialDetailsLists, function (key, item)
                {
                    if (item.ISSUEDATE.trim() == null || item.ISSUEDATE.trim() == "") {
                        item.ISSUEDATE = '  -  ';
                    }

                    if (item.RETURNDATE.trim() == null || item.RETURNDATE.trim() == "") {
                        item.RETURNDATE = '  -  ';

var GetComparisonString = function (ComparisonType, SearchText) {
    var str;

    var searchstr;

    try {
        searchstr = parseInt(SearchText);
    }
    catch (err) {
        searchstr == SearchText;
    }

    switch (ComparisonType) {
        case "contains":
            if (Number.isInteger(searchstr) == true) {
                str = "= " + searchstr;
            }
            else {
                str = " Like N'%" + SearchText + "%' ";
            }
            break;
        case "is equal to":
            if (Number.isInteger(searchstr) == true)
                str = "= " + searchstr;
            else
                str = "LIKE N'" + SearchText + "' ";
            break;

        case "is not equal to":
            if (Number.isInteger(searchstr) == true)
                str = "<> " + searchstr;
            else
                str = "NOT LIKE N'" + SearchText + "' ";
            break;

        case "starts with":
            if (Number.isInteger(searchstr) == true)
                str = "= " + searchstr;
            else
                str = "LIKE N'" + SearchText + "%' ";
            break;

        case "ends with":
            if (Number.isInteger(searchstr) == true)
                str = "= " + searchstr;
            else
                str = "LIKE N'%" + SearchText + "' ";
            break;

        default:
            break;

    }
    return str;
}

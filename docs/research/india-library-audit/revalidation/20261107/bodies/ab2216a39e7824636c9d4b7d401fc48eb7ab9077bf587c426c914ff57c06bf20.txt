function getRelatedData(data,type,prevnext){
	switch(type){
		case 'explore':
			offsetE = getOffset(data,type,prevnext,offsetE,limitE);
			explorer(limitE,offsetE);
			break;
		case 'announce_list':
			offsetA = getOffset(data,type,prevnext,offsetA,limitA);
			getAnnouncementData(limitA,offsetA);
			break;
		case 'partner_lib':
			offsetP = getOffset(data,type,prevnext,offsetP,limitP);
			partnetLibrarirs(limitP,offsetP);
			break;
		case 'accomplish':
			offsetAc = getOffset(data,type,prevnext,offsetAc,limitAc);
			accomplishments(limitAc,offsetAc);
			break;
		case 'featured_books':
			offsetFb = getOffset(data,type,prevnext,offsetFb,limitFb);
			featuredBooks(limitFb,offsetFb);
			break;
		case 'paper_list':
			offsetN = getOffset(data,type,prevnext,offsetN,limitN);
			geteNewsPaper(limitN,offsetN);
			break;
				
	}
}

function getOffset(data,type,prevnext,offset,limit){
	if(prevnext=="next"){
		if(offset >= data.max-limit){
			return offset;
		}else{
			offset = offset + limit
			return offset;
		}
	}else{
		if(offset <= 0){
			return offset;
		}else{
			offset = offset - limit
			return offset;
		}
	}
}

function randomColor() {
	var letters = 'B2D9A'.split('');
	var color = '#';
	for (var i = 0; i < 6; i++ ) {
		color += letters[Math.floor(Math.random() * letters.length)];
	}
	return color;
}

function randomColors() {
	var letters = 'A9D2B'.split('');
	var color = '#';
	for (var i = 0; i < 6; i++ ) {
		color += letters[Math.floor(Math.random() * letters.length)];
	}
	return color;
}
